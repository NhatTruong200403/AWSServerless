Tạo Serverless Hạ tầng

Viết Code Infrastructure as Code (IaC) Với AWS CDK

Bước này, chúng ta sẽ viết mã Infrastructure as Code để triển khai toàn bộ hạ tầng serverless, bao gồm:

  • DynamoDB (on-demand)
  • Lambda Authorizer (JWT custom)
  • Lambda Handler (API logic)
  • HTTP API Gateway với authorizer, caching, usage plan và API key

1. Tạo hoặc chỉnh sửa file lib/aws-serverless-demo-stack.ts

import * as cdk from 'aws-cdk-lib';
import { Construct } from 'constructs';
import * as apigw from 'aws-cdk-lib/aws-apigatewayv2';
import * as integrations from 'aws-cdk-lib/aws-apigatewayv2-integrations';
import * as lambda from 'aws-cdk-lib/aws-lambda-nodejs';
import * as dynamodb from 'aws-cdk-lib/aws-dynamodb';
import * as iam from 'aws-cdk-lib/aws-iam';

export class AwsServerlessDemoStack extends cdk.Stack {
  constructor(scope: Construct, id: string, props?: cdk.StackProps) {
    super(scope, id, props);

    // DynamoDB Table (on-demand cho demo)
    const table = new dynamodb.Table(this, 'DemoItemsTable', {
      partitionKey: { name: 'id', type: dynamodb.AttributeType.STRING },
      billingMode: dynamodb.BillingMode.PAY_PER_REQUEST,
    });

    // Lambda Authorizer (JWT custom, không dùng Cognito)
    const authorizerLambda = new lambda.NodejsFunction(this, 'DemoAuthorizerLambda', {
      entry: 'lambda/authorizer.ts',
      handler: 'handler',
      runtime: lambda.Runtime.NODEJS_20_X,
      environment: {
        JWT_SECRET: 'demo-secret-key-12345', // Thay bằng secret thật cho production
      },
    });

    // Lambda Handler (Logic xử lý GET/POST với validation)
    const handlerLambda = new lambda.NodejsFunction(this, 'DemoHandlerLambda', {
      entry: 'lambda/handler.ts',
      handler: 'handler',
      runtime: lambda.Runtime.NODEJS_20_X,
      environment: {
        TABLE_NAME: table.tableName,
      },
    });
    table.grantReadWrite(handlerLambda);

    // HTTP API Gateway (rẻ hơn REST API)
    const httpApi = new apigw.HttpApi(this, 'DemoHttpApi', {
      apiName: 'ServerlessDemoApi',
    });

    // Authorizer
    const authorizer = new apigw.HttpLambdaAuthorizer('DemoJwtAuthorizer', authorizerLambda, {
      authorizerName: 'DemoJwtAuthorizer',
      identitySource: ['$request.header.Authorization'],
    });

    // Integration
    const lambdaIntegration = new integrations.HttpLambdaIntegration('DemoLambdaIntegration', handlerLambda);

    // Routes (/items GET/POST với auth)
    httpApi.addRoutes({
      path: '/items',
      methods: [apigw.HttpMethod.GET, apigw.HttpMethod.POST],
      integration: lambdaIntegration,
      authorizer: authorizer,
    });

    // Caching cho GET (TTL 60s)
    const stage = httpApi.defaultStage as apigw.HttpStage;
    const cachePolicy = new apigw.CachePolicy(this, 'DemoCachePolicy', {
      defaultTtl: cdk.Duration.seconds(60),
      enableAcceptHeader: true,
    });
    stage.node.addDependency(cachePolicy);

    // Usage Plan + API Key (rate limit: 1000 req/min, quota 10000/day)
    const usagePlan = httpApi.addUsagePlan('DemoUsagePlan', {
      name: 'DemoPlan',
      throttle: { rateLimit: 1000, burstLimit: 2000 },
      quota: { limit: 10000, period: apigw.Period.DAY },
    });

    const apiKey = httpApi.addApiKey('DemoApiKey');
    usagePlan.addApiKey(apiKey);
    usagePlan.addApiStage({ stage: httpApi.defaultStage! });

    // Outputs cho demo
    new cdk.CfnOutput(this, 'ApiUrl', { value: httpApi.url! });
    new cdk.CfnOutput(this, 'ApiKeyId', { value: apiKey.keyId });
  }
}

2. Chỉnh sửa file bin/aws-serverless-demo.ts

import 'source-map-support/register';
import * as cdk from 'aws-cdk-lib';
import { AwsServerlessDemoStack } from '../lib/aws-serverless-demo-stack';

const app = new cdk.App();
new AwsServerlessDemoStack(app, 'AwsServerlessDemoStack', {});

✅ Sau khi hoàn tất, bạn có thể chạy:

cdk synth
cdk deploy

Hệ thống sẽ triển khai toàn bộ hạ tầng serverless trên AWS.
